#define LCD_FILAS 2

// $2  = 2 pulsos
// $5  = 5 pulsos
// $10 = 10 pulsos
//
// El filtro de 70 ms evita rebotes o pulsos duplicados.
// ======================================================

volatile unsigned long pulsos = 0;
volatile unsigned long ultimoPulsoUs = 0;

const unsigned long FILTRO_PULSO_US = 70000;


// ======================================================
// CONFIGURACIÓN WIFI
// ======================================================

const char* ssid = "Starlink D3";

  "https://fichaszd-482614074090.us-central1.run.app/fichas/vender";

const unsigned long HTTP_TIMEOUT_MS = 30000;


// ======================================================
// TIEMPOS DE PANTALLA
// ======================================================
//
// La ficha y el código de reembolso permanecen
// visibles durante 20 segundos.
//
// Si entra una moneda antes, se regresa inmediatamente
// a la pantalla de saldo.
// ======================================================

const unsigned long TIEMPO_MENSAJE_MS = 20000;


// ======================================================
// ESTADOS DE LA APLICACIÓN
// ======================================================

enum EstadoPantalla {
  ESTADO_NORMAL,
  ESTADO_FICHA,
  ESTADO_REEMBOLSO
};

EstadoPantalla estadoPantalla = ESTADO_NORMAL;

unsigned long inicioEstadoPantalla = 0;

// Saldo que existía al mostrar una ficha o reembolso.
// Si cambia, significa que entró una nueva moneda.
unsigned long saldoBasePantalla = 0;


// ======================================================
// PANTALLAS
// ======================================================

// OLED
Adafruit_SSD1306 display(
  SCREEN_WIDTH,
  SCREEN_HEIGHT,
  &Wire,
  -1
);

// LCD 16x2
LiquidCrystal_I2C lcd(
  LCD_ADDRESS,
  LCD_COLUMNAS,
  LCD_FILAS
);


// ======================================================
// INTERRUPCIÓN DEL MONEDERO
// ======================================================
//
// Solo cuenta pulsos.
//
// No utilizar aquí:
// Serial, OLED, LCD, WiFi, HTTP ni delay().
// ======================================================

void IRAM_ATTR contarPulso() {

  unsigned long ahora = micros();
  
  display.println("CONTANDO...");

  if (
    ahora - ultimoPulsoUs >= FILTRO_PULSO_US
  ) {

    pulsos++;

    ultimoPulsoUs = ahora;
  }
}

// ======================================================
// INICIALIZAR OLED + LCD
// ======================================================

void inicializarPantallas() {

  /*
   * OLED y LCD comparten el mismo bus I2C:
   *
   * SDA -> GPIO21
   * SCL -> GPIO22
   */

  Wire.begin(
    I2C_SDA,
    I2C_SCL
  );


  // ==================================================
  // OLED
  // ==================================================

  if (
    !display.begin(
      SSD1306_SWITCHCAPVCC,
      OLED_ADDRESS
    )
  ) {

    Serial.println(
      "No se encontro la pantalla OLED"
    );

    while (true) {
      delay(1000);
    }
  }


  display.setTextColor(
    SSD1306_WHITE
  );

  display.clearDisplay();

  mostrarEncabezadoOLED();

  display.println(
    "Iniciando..."
  );

  display.display();


  // ==================================================
  // LCD
  // ==================================================

  lcd.init();

  lcd.backlight();

  limpiarLCD();

  lcd.print(
    "Zona D CETIS117"
  );

  lcd.setCursor(
    0,
    1
  );

  lcd.print(
    "Iniciando..."
  );
}

  limpiarLCD();

  );


  delay(1000);
}


// ======================================================
// INICIALIZAR MONEDERO
// ======================================================

void inicializarMonedero() {

  pinMode(
    PIN_MONEDERO,
    INPUT_PULLUP
  );


  delay(500);


  attachInterrupt(
    digitalPinToInterrupt(
      PIN_MONEDERO
    ),
    contarPulso,
    FALLING
  );


  Serial.println(
    "Monedero listo"
  );
}

// ======================================================
// EXTRAER FICHA DE LA RESPUESTA JSON
// ======================================================
//
// Ejemplo:
//
// {"Ficha":"211210"}
//
// Resultado:
//
// 211210
// ======================================================

String extraerFicha(
  String json
) {

  int posicionFicha =
    json.indexOf(
      "\"Ficha\""
    );


  if (
    posicionFicha == -1
  ) {

    return "";
  }


  int dosPuntos =
    json.indexOf(
      ":",
      posicionFicha
    );


  int primeraComilla =
    json.indexOf(
      "\"",
      dosPuntos
    );


  int segundaComilla =
    json.indexOf(
      "\"",
      primeraComilla + 1
    );


  if (
    primeraComilla == -1 ||
    segundaComilla == -1
  ) {

    return "";
  }


  return json.substring(
    primeraComilla + 1,
    segundaComilla
  );
}


// ======================================================
// CONSUMIR SERVICIO DE VENTA
// ======================================================

String venderFicha() {

  if (
    WiFi.status() != WL_CONNECTED
  ) {

    Serial.println(
      "WiFi desconectado"
    );

    return "";
  }


  HTTPClient http;


  http.begin(
    URL_VENTA
  );


  http.setTimeout(
    HTTP_TIMEOUT_MS
  );


  http.addHeader(
    "Content-Type",
    "application/json"
  );


  String body =
    "{\"name\":\"Developer\"}";


  Serial.println(
    "Consumiendo servicio de venta..."
  );


  int httpCode =
    http.POST(
      body
    );


  String response = "";


  if (
    httpCode > 0
  ) {

    response =
      http.getString();


    Serial.print(
      "HTTP: "
    );

    Serial.println(
      httpCode
    );


    Serial.print(
      "Respuesta: "
    );

    Serial.println(
      response
    );

  } else {

    Serial.print(
      "Error HTTP: "
    );

    Serial.println(
      http.errorToString(
        httpCode
      )
    );
  }


  http.end();


  return response;
}


// ======================================================
// DESCONTAR PRECIO DE LA OPERACIÓN
// ======================================================
//
// Descuenta únicamente $10.
//
// Si el usuario introdujo más dinero,
// el excedente queda como saldo.
//
// Ejemplo:
//
// $11 - $10 = $1
// ======================================================

unsigned long descontarPrecioFicha() {

  unsigned long saldoRestante;


  noInterrupts();


  if (
    pulsos >= PRECIO_FICHA
  ) {

    pulsos -= PRECIO_FICHA;

  } else {

    pulsos = 0;
  }


  saldoRestante =
    pulsos;


  interrupts();


  return saldoRestante;
}


// ======================================================
// INICIAR PANTALLA TEMPORAL
// ======================================================

void iniciarPantallaTemporal(
  EstadoPantalla nuevoEstado,
  unsigned long saldoActual
) {

  estadoPantalla =
    nuevoEstado;


  saldoBasePantalla =
    saldoActual;


  inicioEstadoPantalla =
    millis();
}


// ======================================================
// PROCESAR PANTALLA TEMPORAL
// ======================================================
//
// Se utiliza tanto para:
// - ficha generada
// - código de reembolso
//
// La pantalla termina cuando:
//
// 1. entra una moneda nueva
// 2. pasan 20 segundos
//
// Al terminar se muestra el saldo disponible
// en OLED y LCD.
// ======================================================

bool procesarPantallaTemporal(
  unsigned long& pulsosAnteriores
) {

  if (
    estadoPantalla == ESTADO_NORMAL
  ) {

    return false;
  }


  unsigned long saldoActual =
    pulsos;


  // ==================================================
  // ENTRÓ UNA NUEVA MONEDA
  // ==================================================

  if (
    saldoActual != saldoBasePantalla
  ) {

    estadoPantalla =
      ESTADO_NORMAL;


    pulsosAnteriores =
      saldoActual;


    Serial.print(
      "Nuevo saldo: $"
    );

    Serial.println(
      saldoActual
    );


    mostrarSaldoActual(
      saldoActual
    );


    return false;
  }


  // ==================================================
  // PASARON 20 SEGUNDOS
  // ==================================================

  if (
    millis() - inicioEstadoPantalla
    >= TIEMPO_MENSAJE_MS
  ) {

    estadoPantalla =
      ESTADO_NORMAL;


    pulsosAnteriores =
      saldoActual;


    mostrarSaldoActual(
      saldoActual
    );


    Serial.print(
      "Pantalla finalizada. Saldo: $"
    );

    Serial.println(
      saldoActual
    );


    return false;
  }


  return true;
}


// ======================================================
// PROCESAR VENTA
// ======================================================
//
// Flujo:
//
// 1. Muestra "Generando ficha".
// 2. Consume Cloud Run una sola vez.
// 3. Descuenta $10.
// 4. Si hay ficha, la muestra.
// 5. Si falla, genera código de reembolso.
// 6. Conserva cualquier saldo superior a $10.
// ======================================================

void procesarVenta(
  unsigned long& pulsosAnteriores
) {

  mostrarGenerandoFicha();


  String respuesta =
    venderFicha();


  String ficha =
    extraerFicha(
      respuesta
    );


  unsigned long saldoRestante =
    descontarPrecioFicha();


  pulsosAnteriores =
    saldoRestante;


  // ==================================================
  // VENTA EXITOSA
  // ==================================================

  if (
    ficha != ""
  ) {

    Serial.print(
      "Ficha generada: "
    );

    Serial.println(
      ficha
    );


    Serial.print(
      "Saldo restante: $"
    );

    Serial.println(
      saldoRestante
    );


    mostrarFicha(
      ficha
    );


    iniciarPantallaTemporal(
      ESTADO_FICHA,
      saldoRestante
    );


    return;
  }


  // ==================================================
  // ERROR / REEMBOLSO
  // ==================================================

  String codigoReembolso =
    generarCodigoReembolso();


  Serial.println(
    "Error al generar ficha"
  );


  Serial.print(
    "Codigo reembolso: "
  );

  Serial.println(
    codigoReembolso
  );


  Serial.print(
    "Saldo restante: $"
  );

  Serial.println(
    saldoRestante
  );


  mostrarCodigoReembolso(
    codigoReembolso
  );


  iniciarPantallaTemporal(
    ESTADO_REEMBOLSO,
    saldoRestante
  );
}


// ======================================================
// SETUP
// ======================================================

void setup() {

  Serial.begin(
    115200
  );


  randomSeed(
    esp_random()
  );


  inicializarPantallas();


  conectarWiFi();


  inicializarMonedero();


  mostrarPantallaInicial();
}


// ======================================================
// LOOP PRINCIPAL
// ======================================================
//
// 1. Controla pantallas temporales.
// 2. Actualiza saldo.
// 3. Al llegar a $10 procesa una venta.
//
// El conteo del dinero ocurre mediante la interrupción
// del GPIO18.
// ======================================================

void loop() {

  static unsigned long pulsosAnteriores = 0;

  static bool procesandoVenta = false;


  // ==================================================
  // FICHA / REEMBOLSO EN PANTALLA
  // ==================================================

  if (
    procesarPantallaTemporal(
      pulsosAnteriores
    )
  ) {

    return;
  }


  unsigned long pulsosActuales =
    pulsos;


  // ==================================================
  // ACTUALIZAR SALDO
  // ==================================================

  if (
    pulsosActuales != pulsosAnteriores
  ) {

    pulsosAnteriores =
      pulsosActuales;


    Serial.print(
      "Saldo acumulado: $"
    );

    Serial.println(
      pulsosActuales
    );


    mostrarSaldoActual(
      pulsosActuales
    );
  }


  // ==================================================
  // GENERAR FICHA
  // ==================================================

  if (
    pulsosActuales >= PRECIO_FICHA
    &&
    !procesandoVenta
  ) {

    procesandoVenta =
      true;


    Serial.print(
      "Saldo disponible: $"
    );

    Serial.println(
      pulsosActuales
    );


    procesarVenta(
      pulsosAnteriores
    );


    procesandoVenta =
      false;


    return;
  }
}
