// ======================================================

// SALDO BASE DEL ERROR

// Saldo que queda después de descontar los $10

//
// saldoBaseError = 1

// pulsos será diferente de saldoBaseError.

);


// ======================================================
// VARIABLES DEL MONEDERO
// ======================================================

volatile unsigned long pulsos = 0;


// ======================================================
// INTERRUPCIÓN DEL MONEDERO
// ======================================================

void IRAM_ATTR contarPulso() {

  unsigned long ahora = micros();

  if (
    ahora - ultimoPulsoUs
    >= FILTRO_PULSO_US
  ) {

    pulsos++;

    ultimoPulsoUs = ahora;
  }
}


// ======================================================
// INICIALIZAR OLED
// ======================================================

void inicializarPantalla() {

  Wire.begin(
    OLED_SDA,
    OLED_SCL
  );


  if (
    !display.begin(
      SSD1306_SWITCHCAPVCC,
      OLED_ADDRESS
    )
  ) {

    Serial.println(
      "No se encontro la pantalla OLED"
    );

    while (true) {

      delay(1000);
    }
  }


  display.clearDisplay();


  display.setTextColor(
    SSD1306_WHITE
  );


  display.setTextSize(2);

  display.setCursor(
    0,
    0
  );


  display.println(
    "Zona D"
  );


  display.setTextSize(1);

  display.println(
    "Cetis 117"
  );


  display.println();


  display.println(
    "Iniciando..."
  );


  display.display();
}


// ======================================================
// CONECTAR WIFI
// ======================================================

void conectarWiFi() {

  display.clearDisplay();

  display.setTextSize(1);

  display.setCursor(
    0,
    0
  );


  display.println(
    "Conectando WiFi..."
  );


  display.display();


  WiFi.begin(
    ssid,
    password
  );


  while (
    WiFi.status()
    != WL_CONNECTED
  ) {

    delay(500);

    Serial.print(".");
  }


  Serial.println();


  Serial.println(
    "WiFi conectado"
  );


  Serial.print(
    "IP: "
  );


  Serial.println(
    WiFi.localIP()
  );


  display.clearDisplay();

  display.setCursor(
    0,
    0
  );


  display.println(
    "WiFi conectado"
  );


  display.display();


  delay(1000);
}


// ======================================================
// CONFIGURAR MONEDERO
// ======================================================

void inicializarMonedero() {

  pinMode(
    PIN_MONEDERO,
    INPUT_PULLUP
  );


  delay(500);


  attachInterrupt(
    digitalPinToInterrupt(
      PIN_MONEDERO
    ),

    contarPulso,

    FALLING
  );


  Serial.println(
    "Monedero listo"
  );


  Serial.println(
    "Esperando pulsos..."
  );
}

// ======================================================
// EXTRAER FICHA DEL JSON
// ======================================================

String extraerFicha(
  String json
) {

  int posicionFicha =
    json.indexOf(
      "\"Ficha\""
    );


  if (
    posicionFicha == -1
  ) {

    return "";
  }


  int dosPuntos =
    json.indexOf(
      ":",
      posicionFicha
    );


  int primeraComilla =
    json.indexOf(
      "\"",
      dosPuntos
    );


  int segundaComilla =
    json.indexOf(
      "\"",
      primeraComilla + 1
    );


  if (
    primeraComilla == -1 ||
    segundaComilla == -1
  ) {

    return "";
  }


  return json.substring(
    primeraComilla + 1,
    segundaComilla
  );
}


// ======================================================
// CONSUMIR SERVICIO REST
// ======================================================

String venderFicha() {

  if (
    WiFi.status()
    != WL_CONNECTED
  ) {

    Serial.println(
      "WiFi desconectado"
    );

    return "";
  }


  HTTPClient http;


  String url =
    "https://fichaszd-482614074090.us-central1.run.app/fichas/vender";


  Serial.println(
    "Consumiendo servicio de venta..."
  );


  http.begin(
    url
  );


  // Tiempo máximo de espera:
  // 30 segundos

  http.setTimeout(
    30000
  );


  http.addHeader(
    "Content-Type",
    "application/json"
  );


  String body =
    "{\"name\":\"Developer\"}";


  int httpCode =
    http.POST(
      body
    );


  String response = "";


  if (
    httpCode > 0
  ) {

    response =
      http.getString();


    Serial.print(
      "Codigo HTTP: "
    );


    Serial.println(
      httpCode
    );


    Serial.println(
      "Respuesta:"
    );


    Serial.println(
      response
    );
  }

  else {

    Serial.print(
      "Error HTTP: "
    );


    Serial.println(
      http.errorToString(
        httpCode
      )
    );
  }


  http.end();


  return response;
}


// ======================================================
// SETUP
// ======================================================

void setup() {

  Serial.begin(
    115200
  );


  randomSeed(
    esp_random()
  );


  inicializarPantalla();


  conectarWiFi();


  inicializarMonedero();


  mostrarPantallaInicial();
}


// ======================================================
// LOOP
// ======================================================

void loop() {

  static unsigned long pulsosAnteriores = 0;

  static bool procesandoVenta = false;


  unsigned long pulsosActuales =
    pulsos;


  // ==================================================
  // MOSTRANDO ERROR / CÓDIGO DE REEMBOLSO
  // ==================================================

  if (
    mostrandoError
  ) {

    // Volvemos a leer el saldo actual.
    //
    // Puede haber cambiado porque las interrupciones
    // siguen contando monedas aunque estemos mostrando
    // el código de reembolso.

    pulsosActuales =
      pulsos;


    // ==================================================
    // ENTRÓ UNA NUEVA MONEDA DURANTE EL ERROR
    // ==================================================
    //
    // saldoBaseError contiene el saldo que quedó
    // después de descontar los $10.
    //
    // Si pulsos cambia significa que entró
    // dinero adicional.
    // ==================================================

    if (
      pulsosActuales
      != saldoBaseError
    ) {

      mostrandoError =
        false;


      pulsosAnteriores =
        pulsosActuales;


      Serial.println(
        "Nueva moneda detectada durante reembolso."
      );


      Serial.print(
        "Saldo actual: $"
      );


      Serial.println(
        pulsosActuales
      );


      mostrarSaldo(
        pulsosActuales
      );


      return;
    }


    // ==================================================
    // PASARON 20 SEGUNDOS
    // ==================================================

    if (
      millis() - inicioPantallaError
      >= TIEMPO_ERROR_MS
    ) {

      mostrandoError =
        false;


      pulsosActuales =
        pulsos;


      pulsosAnteriores =
        pulsosActuales;


      Serial.println(
        "Pantalla de reembolso finalizada."
      );


      // --------------------------------------------------
      // HAY SALDO A FAVOR
      // --------------------------------------------------

      if (
        pulsosActuales > 0
      ) {

        Serial.print(
          "Saldo a favor: $"
        );


        Serial.println(
          pulsosActuales
        );


        mostrarSaldo(
          pulsosActuales
        );
      }


      // --------------------------------------------------
      // NO HAY SALDO
      // --------------------------------------------------

      else {

        mostrarPantallaInicial();


        Serial.println(
          "Saldo: $0"
        );
      }


      return;
    }


    // Mientras mostramos el código de reembolso
    // no ejecutamos ninguna otra lógica.

    return;
  }


  // ==================================================
  // ESTAMOS MOSTRANDO UNA FICHA EXITOSA
  // ==================================================

  if (
    mostrandoFicha
  ) {

    pulsosActuales =
      pulsos;


    // ==================================================
    // ENTRÓ UNA NUEVA MONEDA
    // ==================================================

    if (
      pulsosActuales
      != saldoBaseFicha
    ) {

      mostrandoFicha =
        false;


      pulsosAnteriores =
        pulsosActuales;


      Serial.println(
        "Nueva moneda detectada."
      );


      Serial.print(
        "Saldo actual: $"
      );


      Serial.println(
        pulsosActuales
      );


      mostrarSaldo(
        pulsosActuales
      );


      return;
    }


    // ==================================================
    // PASARON 20 SEGUNDOS
    // ==================================================

    if (
      millis() - inicioPantallaFicha
      >= TIEMPO_FICHA_MS
    ) {

      mostrandoFicha =
        false;


      pulsosActuales =
        pulsos;


      pulsosAnteriores =
        pulsosActuales;


      Serial.println(
        "Ficha finalizada."
      );


      // --------------------------------------------------
      // HAY SALDO A FAVOR
      // --------------------------------------------------

      if (
        pulsosActuales > 0
      ) {

        Serial.print(
          "Saldo a favor: $"
        );


        Serial.println(
          pulsosActuales
        );


        mostrarSaldo(
          pulsosActuales
        );
      }


      // --------------------------------------------------
      // NO HAY SALDO
      // --------------------------------------------------

      else {

        mostrarPantallaInicial();


        Serial.println(
          "Saldo: $0"
        );
      }


      return;
    }


    return;
  }


  // ==================================================
  // ACTUALIZAR SALDO NORMAL
  // ==================================================

  if (
    pulsosActuales
    != pulsosAnteriores
  ) {

    pulsosAnteriores =
      pulsosActuales;


    Serial.print(
      "Saldo acumulado: $"
    );


    Serial.println(
      pulsosActuales
    );


    mostrarSaldo(
      pulsosActuales
    );
  }


  // ==================================================
  // GENERAR FICHA CUANDO SALDO >= $10
  // ==================================================

  if (
    pulsosActuales >= PRECIO_FICHA
    &&
    !procesandoVenta
  ) {

    procesandoVenta =
      true;


    Serial.print(
      "Saldo disponible: $"
    );


    Serial.println(
      pulsosActuales
    );


    Serial.println(
      "Generando ficha..."
    );


    // ==================================================
    // MOSTRAR MENSAJE DE ESPERA
    // ==================================================

    mostrarGenerandoFicha();


    // ==================================================
    // CONSUMIR SERVICIO UNA SOLA VEZ
    // ==================================================

    String respuesta =
      venderFicha();


    Serial.println(
      "Respuesta servicio:"
    );


    Serial.println(
      respuesta
    );


    // ==================================================
    // EXTRAER FICHA
    // ==================================================

    String ficha =
      extraerFicha(
        respuesta
      );


    // ==================================================
    // VENTA EXITOSA
    // ==================================================

    if (
      ficha != ""
    ) {

      Serial.print(
        "Ficha generada: "
      );


      Serial.println(
        ficha
      );


      // ==================================================
      // DESCONTAR SOLAMENTE $10
      // ==================================================

      noInterrupts();


      if (
        pulsos >= PRECIO_FICHA
      ) {

        pulsos =
          pulsos - PRECIO_FICHA;
      }

      else {

        pulsos = 0;
      }


      // Guardamos el saldo restante
      // justo después de la venta.

      saldoBaseFicha =
        pulsos;


      interrupts();


      pulsosAnteriores =
        saldoBaseFicha;


      Serial.print(
        "Saldo restante: $"
      );


      Serial.println(
        saldoBaseFicha
      );


      // ==================================================
      // MOSTRAR FICHA
      // ==================================================

      mostrarFicha(
        ficha
      );


      mostrandoFicha =
        true;


      inicioPantallaFicha =
        millis();


      procesandoVenta =
        false;


      return;
    }


    // ==================================================
    // ERROR AL GENERAR FICHA
    // ==================================================
    //
    // IMPORTANTE:
    //
    // Los primeros $10 quedan asociados al
    // código de reembolso.
    //
    // Solamente conservamos como saldo
    // cualquier cantidad superior a $10.
    //
    // Ejemplo:
    //
    // $11
    //
    // Código reembolso -> $10
    // Saldo a favor    -> $1
    //
    // ==================================================

    Serial.println(
      "Error al generar ficha"
    );


    // ==================================================
    // DESCONTAR LOS $10 DEL CÓDIGO DE REEMBOLSO
    // ==================================================

    noInterrupts();


    if (
      pulsos >= PRECIO_FICHA
    ) {

      pulsos =
        pulsos - PRECIO_FICHA;
    }

    else {

      pulsos = 0;
    }


    // Guardamos el saldo restante.

    saldoBaseError =
      pulsos;


    interrupts();


    pulsosAnteriores =
      saldoBaseError;


    Serial.print(
      "Saldo restante despues del error: $"
    );


    Serial.println(
      saldoBaseError
    );


    // ==================================================
    // GENERAR CÓDIGO DE REEMBOLSO
    // ==================================================

    String codigoReembolso =
      generarCodigoReembolso();


    Serial.print(
      "Codigo reembolso: "
    );


    Serial.println(
      codigoReembolso
    );


    // ==================================================
    // MOSTRAR CÓDIGO
    // ==================================================

    mostrarErrorReembolso(
      codigoReembolso
    );


    mostrandoError =
      true;


    inicioPantallaError =
      millis();


    procesandoVenta =
      false;


    return;
  }
}
