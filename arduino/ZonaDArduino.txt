#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <WiFi.h>
#include <HTTPClient.h>


// ======================================================
// CONFIGURACIÓN GENERAL
// ======================================================

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64

#define PIN_MONEDERO 18

#define OLED_SDA 21
#define OLED_SCL 22
#define OLED_ADDRESS 0x3C

const unsigned long PRECIO_FICHA = 10;


// ======================================================
// CONFIGURACIÓN DEL MONEDERO
// ======================================================
//
// El monedero genera:
// $1  = 1 pulso
// $2  = 2 pulsos
// $5  = 5 pulsos
// $10 = 10 pulsos
//
// Se utiliza un filtro de 70 ms para evitar
// rebotes o pulsos duplicados.
// ======================================================

volatile unsigned long pulsos = 0;
volatile unsigned long ultimoPulsoUs = 0;

const unsigned long FILTRO_PULSO_US = 70000;


// ======================================================
// CONFIGURACIÓN WIFI
// ======================================================

const char* ssid = "Zona-D";

// Coloca aquí tu contraseña WiFi
const char* password = "147896325d";


// ======================================================
// CONFIGURACIÓN DEL SERVICIO REST
// ======================================================

const char* URL_VENTA =
  "https://fichaszd-482614074090.us-central1.run.app/fichas/vender";

const unsigned long HTTP_TIMEOUT_MS = 30000;


// ======================================================
// TIEMPOS DE PANTALLA
// ======================================================
//
// Tanto una ficha generada como un código de reembolso
// permanecen visibles durante 20 segundos.
//
// Si entra una nueva moneda antes de ese tiempo,
// inmediatamente se vuelve a mostrar el saldo.
// ======================================================

const unsigned long TIEMPO_MENSAJE_MS = 20000;


// ======================================================
// ESTADOS DE LA PANTALLA
// ======================================================
//
// NORMAL:
//   Muestra el saldo y permite generar una ficha.
//
// FICHA:
//   Muestra la ficha generada.
//
// REEMBOLSO:
//   Muestra el código generado cuando falla la venta.
// ======================================================

enum EstadoPantalla {
  ESTADO_NORMAL,
  ESTADO_FICHA,
  ESTADO_REEMBOLSO
};

EstadoPantalla estadoPantalla = ESTADO_NORMAL;

unsigned long inicioEstadoPantalla = 0;

// Saldo existente cuando comenzó una pantalla temporal.
// Permite detectar si entró una moneda nueva.
unsigned long saldoBasePantalla = 0;


// ======================================================
// OLED
// ======================================================

Adafruit_SSD1306 display(
  SCREEN_WIDTH,
  SCREEN_HEIGHT,
  &Wire,
  -1
);


// ======================================================
// INTERRUPCIÓN DEL MONEDERO
// ======================================================
//
// Esta función se ejecuta automáticamente cada vez
// que GPIO18 detecta un pulso del PC817.
//
// Debe ser rápida:
// no usar Serial, WiFi, HTTP, display ni delay() aquí.
// ======================================================

void IRAM_ATTR contarPulso() {

  unsigned long ahora = micros();

  if (
    ahora - ultimoPulsoUs >= FILTRO_PULSO_US
  ) {

    pulsos++;
    ultimoPulsoUs = ahora;
  }
}


// ======================================================
// FUNCIONES AUXILIARES DE PANTALLA
// ======================================================

// Encabezado común utilizado en las pantallas.
void mostrarEncabezado() {

  display.setTextSize(2);
  display.setCursor(0, 0);
  display.println("Zona D");

  display.setTextSize(1);
  display.println("Cetis 117");
  display.println();
}


// Pantalla principal cuando no existe saldo.
void mostrarPantallaInicial() {

  display.clearDisplay();

  mostrarEncabezado();

  display.println("Saldo:");

  display.setTextSize(2);
  display.println("$0");

  display.display();
}


// Muestra el saldo disponible.
void mostrarSaldo(
  unsigned long saldo
) {

  display.clearDisplay();

  mostrarEncabezado();

  display.println("Saldo acumulado:");

  display.setTextSize(3);
  display.print("$");
  display.println(saldo);

  display.display();
}


// Decide automáticamente si mostrar $0
// o un saldo existente.
void mostrarSaldoActual(
  unsigned long saldo
) {

  if (saldo > 0) {

    mostrarSaldo(saldo);

  } else {

    mostrarPantallaInicial();
  }
}


// Muestra la ficha obtenida del servicio.
void mostrarFicha(
  String ficha
) {

  display.clearDisplay();

  mostrarEncabezado();

  display.println("Ficha Generada:");

  display.setTextSize(3);
  display.println(ficha);

  display.display();
}


// Mensaje mostrado mientras se realiza
// la petición HTTP.
void mostrarGenerandoFicha() {

  display.clearDisplay();

  mostrarEncabezado();

  display.println("Generando ficha,");
  display.println("espere un momento");
  display.println("por favor...");

  display.display();
}


// Muestra el código generado cuando
// no fue posible obtener una ficha.
void mostrarCodigoReembolso(
  String codigo
) {

  display.clearDisplay();

  mostrarEncabezado();

  display.println("Codigo de Reembolso:");
  display.println();

  display.setTextSize(2);
  display.println(codigo);

  display.display();
}


// ======================================================
// INICIALIZAR OLED
// ======================================================

void inicializarPantalla() {

  Wire.begin(
    OLED_SDA,
    OLED_SCL
  );

  if (
    !display.begin(
      SSD1306_SWITCHCAPVCC,
      OLED_ADDRESS
    )
  ) {

    Serial.println(
      "No se encontro la pantalla OLED"
    );

    while (true) {
      delay(1000);
    }
  }

  display.setTextColor(
    SSD1306_WHITE
  );

  display.clearDisplay();

  mostrarEncabezado();

  display.println(
    "Iniciando..."
  );

  display.display();
}


// ======================================================
// CONECTAR WIFI
// ======================================================

void conectarWiFi() {

  display.clearDisplay();

  display.setCursor(0, 0);
  
  mostrarEncabezado();

  display.setTextSize(1);
  display.println(
    "Conectando a Zona D.."
  );

  display.display();


  WiFi.begin(
    ssid,
    password
  );


  while (
    WiFi.status() != WL_CONNECTED
  ) {

    delay(500);
    Serial.print(".");
  }


  Serial.println();
  Serial.println(
    "WiFi conectado"
  );

  Serial.print(
    "IP: "
  );

  Serial.println(
    WiFi.localIP()
  );


  display.clearDisplay();
  display.setCursor(0, 0);

  display.println(
    "Zona D conectado"
  );

  display.display();

  delay(1000);
}


// ======================================================
// INICIALIZAR MONEDERO
// ======================================================

void inicializarMonedero() {

  pinMode(
    PIN_MONEDERO,
    INPUT_PULLUP
  );

  delay(500);

  attachInterrupt(
    digitalPinToInterrupt(
      PIN_MONEDERO
    ),
    contarPulso,
    FALLING
  );

  Serial.println(
    "Monedero listo"
  );
}


// ======================================================
// GENERAR CÓDIGO DE REEMBOLSO
// ======================================================
//
// El código se utiliza como referencia para devolver
// los $10 correspondientes a una operación fallida.
// ======================================================

String generarCodigoReembolso() {

  unsigned long numero =
    random(
      100000,
      1000000
    );

  return
    "ZD-" + String(numero);
}


// ======================================================
// EXTRAER FICHA DE LA RESPUESTA JSON
// ======================================================
//
// Espera algo como:
//
// {"Ficha":"211210"}
//
// Devuelve:
//
// 211210
//
// Si "Ficha" no existe, devuelve cadena vacía.
// ======================================================

String extraerFicha(
  String json
) {

  int posicionFicha =
    json.indexOf(
      "\"Ficha\""
    );

  if (
    posicionFicha == -1
  ) {

    return "";
  }


  int dosPuntos =
    json.indexOf(
      ":",
      posicionFicha
    );

  int primeraComilla =
    json.indexOf(
      "\"",
      dosPuntos
    );

  int segundaComilla =
    json.indexOf(
      "\"",
      primeraComilla + 1
    );


  if (
    primeraComilla == -1 ||
    segundaComilla == -1
  ) {

    return "";
  }


  return json.substring(
    primeraComilla + 1,
    segundaComilla
  );
}


// ======================================================
// CONSUMIR SERVICIO DE VENTA
// ======================================================
//
// Realiza una única petición POST.
//
// Si el servicio responde correctamente,
// devuelve el JSON recibido.
//
// Si ocurre un error de red o WiFi,
// devuelve cadena vacía.
// ======================================================

String venderFicha() {

  if (
    WiFi.status() != WL_CONNECTED
  ) {

    Serial.println(
      "WiFi desconectado"
    );

    return "";
  }


  HTTPClient http;

  http.begin(
    URL_VENTA
  );

  http.setTimeout(
    HTTP_TIMEOUT_MS
  );

  http.addHeader(
    "Content-Type",
    "application/json"
  );


  String body =
    "{\"name\":\"Developer\"}";


  Serial.println(
    "Consumiendo servicio de venta..."
  );


  int httpCode =
    http.POST(
      body
    );


  String response = "";


  if (
    httpCode > 0
  ) {

    response =
      http.getString();

    Serial.print(
      "HTTP: "
    );

    Serial.println(
      httpCode
    );

    Serial.print(
      "Respuesta: "
    );

    Serial.println(
      response
    );

  } else {

    Serial.print(
      "Error HTTP: "
    );

    Serial.println(
      http.errorToString(
        httpCode
      )
    );
  }


  http.end();

  return response;
}


// ======================================================
// DESCONTAR EL PRECIO DE UNA OPERACIÓN
// ======================================================
//
// Tanto una ficha exitosa como un reembolso
// consumen únicamente $10 del saldo.
//
// Cualquier cantidad superior queda disponible.
//
// Ejemplo:
//
// saldo = $11
// operación = $10
// saldo restante = $1
//
// Se deshabilitan momentáneamente las interrupciones
// para modificar el contador de forma segura.
// ======================================================

unsigned long descontarPrecioFicha() {

  unsigned long saldoRestante;


  noInterrupts();


  if (
    pulsos >= PRECIO_FICHA
  ) {

    pulsos -= PRECIO_FICHA;

  } else {

    pulsos = 0;
  }


  saldoRestante =
    pulsos;


  interrupts();


  return saldoRestante;
}


// ======================================================
// INICIAR PANTALLA TEMPORAL
// ======================================================
//
// Registra:
// - qué pantalla estamos mostrando
// - cuándo comenzó
// - qué saldo existía en ese momento
//
// Si el saldo cambia significa que entró
// una moneda nueva.
// ======================================================

void iniciarPantallaTemporal(
  EstadoPantalla nuevoEstado,
  unsigned long saldoActual
) {

  estadoPantalla =
    nuevoEstado;

  saldoBasePantalla =
    saldoActual;

  inicioEstadoPantalla =
    millis();
}


// ======================================================
// PROCESAR PANTALLA TEMPORAL
// ======================================================
//
// Se utiliza tanto para:
// - ficha generada
// - código de reembolso
//
// Devuelve true mientras la pantalla temporal
// está activa.
//
// La pantalla termina cuando:
// 1. entra una nueva moneda
// 2. transcurren 20 segundos
// ======================================================

bool procesarPantallaTemporal(
  unsigned long& pulsosAnteriores
) {

  if (
    estadoPantalla == ESTADO_NORMAL
  ) {

    return false;
  }


  unsigned long saldoActual =
    pulsos;


  // --------------------------------------------------
  // Entró una moneda nueva
  // --------------------------------------------------

  if (
    saldoActual != saldoBasePantalla
  ) {

    estadoPantalla =
      ESTADO_NORMAL;

    pulsosAnteriores =
      saldoActual;


    Serial.print(
      "Nuevo saldo: $"
    );

    Serial.println(
      saldoActual
    );


    mostrarSaldoActual(
      saldoActual
    );


    return false;
  }


  // --------------------------------------------------
  // Transcurrieron los 20 segundos
  // --------------------------------------------------

  if (
    millis() - inicioEstadoPantalla
    >= TIEMPO_MENSAJE_MS
  ) {

    estadoPantalla =
      ESTADO_NORMAL;

    pulsosAnteriores =
      saldoActual;


    mostrarSaldoActual(
      saldoActual
    );


    Serial.print(
      "Pantalla finalizada. Saldo: $"
    );

    Serial.println(
      saldoActual
    );


    return false;
  }


  return true;
}


// ======================================================
// PROCESAR VENTA
// ======================================================
//
// Flujo:
//
// 1. muestra "Generando ficha"
// 2. consume Cloud Run una sola vez
// 3. si obtiene ficha:
//      - descuenta $10
//      - conserva excedente
//      - muestra ficha
//
// 4. si falla:
//      - descuenta $10
//      - conserva excedente
//      - genera código de reembolso
// ======================================================

void procesarVenta(
  unsigned long& pulsosAnteriores
) {

  mostrarGenerandoFicha();


  String respuesta =
    venderFicha();


  String ficha =
    extraerFicha(
      respuesta
    );


  unsigned long saldoRestante =
    descontarPrecioFicha();


  pulsosAnteriores =
    saldoRestante;


  // ==================================================
  // VENTA EXITOSA
  // ==================================================

  if (
    ficha != ""
  ) {

    Serial.print(
      "Ficha generada: "
    );

    Serial.println(
      ficha
    );


    Serial.print(
      "Saldo restante: $"
    );

    Serial.println(
      saldoRestante
    );


    mostrarFicha(
      ficha
    );


    iniciarPantallaTemporal(
      ESTADO_FICHA,
      saldoRestante
    );


    return;
  }


  // ==================================================
  // ERROR / REEMBOLSO
  // ==================================================

  String codigoReembolso =
    generarCodigoReembolso();


  Serial.println(
    "Error al generar ficha"
  );


  Serial.print(
    "Codigo reembolso: "
  );

  Serial.println(
    codigoReembolso
  );


  Serial.print(
    "Saldo restante: $"
  );

  Serial.println(
    saldoRestante
  );


  mostrarCodigoReembolso(
    codigoReembolso
  );


  iniciarPantallaTemporal(
    ESTADO_REEMBOLSO,
    saldoRestante
  );
}


// ======================================================
// SETUP
// ======================================================

void setup() {

  Serial.begin(
    115200
  );


  randomSeed(
    esp_random()
  );


  inicializarPantalla();

  conectarWiFi();

  inicializarMonedero();

  mostrarPantallaInicial();
}


// ======================================================
// LOOP PRINCIPAL
// ======================================================
//
// El loop tiene únicamente tres responsabilidades:
//
// 1. mantener las pantallas temporales
// 2. actualizar el saldo
// 3. iniciar una venta al llegar a $10
//
// El conteo de monedas ocurre de forma independiente
// mediante la interrupción contarPulso().
// ======================================================

void loop() {

  static unsigned long pulsosAnteriores = 0;

  static bool procesandoVenta = false;


  // ==================================================
  // PANTALLA DE FICHA / REEMBOLSO
  // ==================================================

  if (
    procesarPantallaTemporal(
      pulsosAnteriores
    )
  ) {

    return;
  }


  unsigned long pulsosActuales =
    pulsos;


  // ==================================================
  // ACTUALIZAR SALDO
  // ==================================================

  if (
    pulsosActuales != pulsosAnteriores
  ) {

    pulsosAnteriores =
      pulsosActuales;


    Serial.print(
      "Saldo acumulado: $"
    );

    Serial.println(
      pulsosActuales
    );


    mostrarSaldoActual(
      pulsosActuales
    );
  }


  // ==================================================
  // GENERAR FICHA
  // ==================================================

  if (
    pulsosActuales >= PRECIO_FICHA
    &&
    !procesandoVenta
  ) {

    procesandoVenta =
      true;


    Serial.print(
      "Saldo disponible: $"
    );

    Serial.println(
      pulsosActuales
    );


    procesarVenta(
      pulsosAnteriores
    );


    procesandoVenta =
      false;


    return;
  }
}
